This chapter describes the implementation details of the GroDealz platform. It outlines the technology stack, walks through the core backend services, and details the frontend component architecture.

\section{Technology Stack}

The implementation utilizes an integrated, multi-service architecture:

\begin{itemize}
    \item \textbf{Frontend:} React 18 with functional components and hooks (\texttt{useState}, \texttt{useEffect}, \texttt{useContext}, \texttt{useMemo}). Routing is managed by React Router v6. Styling is built using custom CSS modules with dark-mode design tokens, ensuring responsive layouts without external heavyweight UI frameworks.
    \item \textbf{Backend Core (REST and Real-Time):} Node.js runtime with Express 4 web framework. Bidirectional communication is implemented using Socket.IO v4. Background scheduled tasks (such as hourly demand forecasts) are executed via \texttt{node-cron}. Security is enforced using \texttt{bcryptjs} for password hashing, \texttt{jsonwebtoken} for token creation and validation, \texttt{cors} for origin management, and \texttt{express-rate-limit} for request throttling.
    \item \textbf{Database and Persistence:} MongoDB Atlas cloud database managed via the Mongoose 7 Object Document Mapper (ODM). Indexes are defined on frequent query paths (such as \texttt{requestId}, \texttt{customer}, \texttt{store}, and \texttt{status}).
    \item \textbf{Autonomous AI Agent Service:} Built using Python 3.10 with FastAPI and LangGraph. It models multi-step reasoning nodes for cross-store catalog searching and basket cost optimization.
\end{itemize}

\section{Core Backend Services}

\subsection{Natural Language Requirement Parser}
The requirement parser takes arbitrary customer sentences (such as \textbf{``I need 25 kg basmati rice for a hostel. My budget is ₹2,500. Delivery within two days.''}) and extracts structured parameters using regular expressions and semantic fallback dictionaries:

\begin{itemize}
    \item \textbf{Budget Extraction:} Strips currency symbols (₹, Rs, INR) and comma digit separators, matching numerical patterns adjacent to budget keywords (e.g., \textbf{``budget is ₹2,500''}, \textbf{``under ₹600''}).
    \item \textbf{Delivery Deadline Extraction:} Scans for day counts (e.g., \textbf{``within 2 days''}, \textbf{``in 3 days''}) or relative terms (such as \textbf{``tomorrow''} or \textbf{``urgent''}) and computes the future \texttt{Date} timestamp.
    \item \textbf{Commodity and Quantity Extraction:} Matches patterns pairing numbers with measurement units (kg, g, L, packs, pcs) and product names (such as \textbf{``25 kg basmati rice''} or \textbf{``2 litre mustard oil''}).
\end{itemize}

Algorithm~\ref{alg:rfq_parser} specifies the complete multi-pattern entity extraction and normalization pipeline:

\begin{algorithm}[H]
\caption{Natural Language Shopping Requirement Parser (\texttt{rfqParser.js})}
\label{alg:rfq_parser}
\begin{algorithmic}[1]
\Require Raw natural language prompt $Q$, Current system timestamp $t_{\text{now}}$
\Ensure Structured requirement object $R = \langle B, t_{\text{deadline}}, I \rangle$
\State $Q_{\text{clean}} \gets \textsc{NormalizeCommasAndWhitespace}(Q)$
\State $B \gets \textsc{ExtractBudgetPattern}(Q_{\text{clean}})$ \Comment{Regex matching for budget keywords and currency}
\State $t_{\text{deadline}} \gets \textsc{ExtractDeadline}(Q_{\text{clean}}, t_{\text{now}})$ \Comment{Computes future delivery date}
\State $I \gets \emptyset$
\For{each commodity match $m$ in $\textsc{TokenizeCommodityPatterns}(Q_{\text{clean}})$}
    \State $(q, u) \gets \textsc{ExtractQuantityAndUnit}(m)$ \Comment{e.g., $25\text{ kg}$, $2\text{ litre}$}
    \State $p \gets \textsc{ExtractProductName}(m)$ \Comment{e.g., ``basmati rice''}
    \State $I \gets I \cup \{\langle \text{name}: p, \text{quantity}: q, \text{unit}: u \rangle\}$
\EndFor
\State \Return $R \gets \langle \text{budget}: B, \text{deadline}: t_{\text{deadline}}, \text{items}: I \rangle$
\end{algorithmic}
\end{algorithm}

\subsection{Merchant Quotation Engine}
The quotation engine governs quotation validation, multi-criteria dynamic ranking, negotiation state management, and atomic conversion to confirmed orders. Algorithm~\ref{alg:quotation_ranking} formalizes the line-item validation and scoring procedure:

\begin{algorithm}[H]
\caption{Itemized Quotation Validation and Multi-Criteria Dynamic Ranking}
\label{alg:quotation_ranking}
\begin{algorithmic}[1]
\Require Quoted line items $L$, delivery fee $F$, tax rate $\tau$, customer priority $M$, target budget $B_{\text{target}}$
\Ensure Validated quotation record $Q$ with dynamic ranking score $S$
\State $\text{subtotal} \gets 0$
\For{each line item $i \in L$}
    \State $\text{lineTotal}_i \gets \text{round}(\text{qty}_i \times \text{unitPrice}_i, 2)$
    \State $\text{subtotal} \gets \text{subtotal} + \text{lineTotal}_i$
\EndFor
\State $\text{grandTotal} \gets \text{round}(\text{subtotal} + F + (\text{subtotal} \times \tau), 2)$
\If{$M = \text{\texttt{lowest\_price}}$}
    \State $S \gets 1{,}000{,}000 - \text{grandTotal}$
\ElsIf{$M = \text{\texttt{fastest\_delivery}}$}
    \State $S \gets T_{\text{future}} - t_{\text{delivery}}$
\ElsIf{$M = \text{\texttt{highest\_rated}}$}
    \State $S \gets (\text{Rating} \times 1{,}000) + \min(\text{ReviewCount}, 100)$
\ElsIf{$M = \text{\texttt{balanced}}$}
    \State $S_{\text{price}} \gets 50 \times \left(1 - \frac{\text{grandTotal} - P_{\min}}{P_{\max} - P_{\min}}\right)$
    \State $S_{\text{speed}} \gets 30 \times \left(1 - \frac{t_{\text{delivery}} - t_{\text{earliest}}}{t_{\text{latest}} - t_{\text{earliest}}}\right)$
    \State $S_{\text{rating}} \gets 20 \times \left(\frac{\text{Rating}}{5.0}\right)$
    \State $S \gets S_{\text{price}} + S_{\text{speed}} + S_{\text{rating}} + (\text{grandTotal} \le B_{\text{target}} \ ? \ 5 : 0)$
\EndIf
\State \Return $Q \gets \langle \text{lineItems}: L, \text{grandTotal}: \text{grandTotal}, \text{score}: S \rangle$
\end{algorithmic}
\end{algorithm}

The quotation engine also manages negotiation state. When a customer counters, the quote status updates to \texttt{customer\_countered}. When a merchant revises, the previous version is archived in \texttt{revisionHistory} and new totals are calculated. Once accepted, the quotation is converted into an active order.

\subsection{Hybrid Recommendation Subsystem}
The recommendation service in \texttt{backend/routes/recommendations.js} calculates scores using a three-tier model:
\begin{enumerate}
    \item \textbf{Collaborative Filtering:} Computes cosine similarity between user interaction history vectors. If a user is new (cold-start), the engine falls back to trending items sorted by total order count.
    \item \textbf{Content-Based Scoring:} Evaluates 16-dimensional feature vectors representing product categories, dietary tags, and price points.
    \item \textbf{Context-Aware Scoring:} Analyzes local time of day to apply category multipliers (+0.10 for Dairy and Bakery in morning hours; +0.12 for Vegetables and Staples in evening hours).
\end{enumerate}

\subsection{Automated Demand Prediction}
Implemented in \texttt{backend/ai/demandPrediction.js}, an automated \texttt{node-cron} job triggers hourly. It aggregates past order volumes over a rolling 168-hour window, applies single exponential smoothing ($\alpha = 0.4$), computes moving-average trend direction, and calculates multivariate regression to output predicted demand scores and staffing recommendations.

\section{Frontend Architecture and Components}

The user interface is organized into modular pages located in \texttt{frontend/src/pages/}:

\subsection{Customer Shopping Requirements Page}
\begin{itemize}
    \item \textbf{Prompt Input with AI Extractor:} Features an interactive input box with a 1-click example loader (\textbf{``I need 25 kg basmati rice for a hostel...''}). Clicking \textbf{``Extract with AI Parser''} calls the backend API and auto-fills the line-item table.
    \item \textbf{Structured Line-Item Editor:} Allows users to add, edit, or delete items, specify units, and record preferred brands.
    \item \textbf{Offer Comparison Matrix:} Displays competing merchant cards side-by-side. Users can switch ranking priorities dynamically with one click, expand line-item breakdowns to view unit prices, propose counter-offers, or accept the winning quote to generate an immediate order.
\end{itemize}

\subsection{Merchant Quotation Dashboard}
\begin{itemize}
    \item \textbf{Open Buyer Requests Feed:} Displays open nearby requests with item details, delivery location, deadline countdown, and budget visibility indicator.
    \item \textbf{Itemized Quotation Modal:} Pre-fills requested items so the merchant can enter individual unit prices, set delivery fees and taxes, confirm inventory stock, choose delivery dates, and set quote validity periods.
    \item \textbf{My Bids and Counter Negotiation View:} Highlights active bids. When a customer sends a counter-offer, an amber alert banner displays the proposed price and note, with a 1-click option to revise the quotation.
\end{itemize}

\subsection{Autonomous AI Grocery Planner}
Provides the conversational grocery interface with parameter chips (Family Size, Duration, Budget, Diet, Goal), a visual Store Journey Audit showing the picking route, and a comparative savings banner highlighting the multi-store net advantage over single-store purchases.

\subsection{Customer Support CRM}
Allows customers to create tickets with an optional PII masking toggle. It features automated intent classification, AI confidence percentage ratings, CRM tracking identifiers, and a live chat panel.

\subsection{Homepage and Product Catalog}
Presents the full grocery catalog with category pills, live search, freshness badges, and personalized recommendation rails.

\section{Autonomous LangGraph Decision Graph Implementation}

The autonomous shopping assistant is implemented in the Python microservice (\texttt{ai-agent/app/graph.py}) using LangGraph. Unlike rigid conversational scripts, the agent maintains an explicit, typed state graph throughout its multi-step execution.

\subsection{State Schema Specification}
The workflow state is defined as an immutable data contract passing typed state channels between decision nodes, as detailed in Table~\ref{tab:agent_state}.

\begin{table}[H]
\centering
\caption{Specification of LangGraph Workflow State Channels (\texttt{AgentState})}
\label{tab:agent_state}
\begin{tabular}{>{\raggedright\arraybackslash}p{4.45cm} >{\raggedright\arraybackslash}p{2.1cm} >{\raggedright\arraybackslash}p{7.45cm}}
\toprule
\textbf{State Channel} & \textbf{Data Type} & \textbf{Operational Function} \\

\subsection{Graph Node Pipeline and Execution Flow}
The decision graph executes six sequential operational nodes:
\begin{enumerate}
    \item \textbf{Requirement Extraction Node (extract\_requirements\_node):} Invokes an LLM prompt with strict Pydantic output parsing (\texttt{ShoppingRequirements}) to extract household size, planning duration, target budget, and item lists.
    \item \textbf{Sanity Validation Node (validate\_requirements\_node):} Performs sanity validation. If the customer submits an open-ended goal (e.g., \textbf{``Groceries for 3 days''}) without specific items, it dynamically injects baseline staples (vegetables, rice, milk, dal, bread) scaled by family size.
    \item \textbf{Multi-Store Search Node (multi\_store\_search\_node):} Iterates through needed items and calls \texttt{store\_registry.parallel\_search()}, executing concurrent asynchronous HTTP requests across all registered store adapters (FreshMart Central, QuickBazaar, Paona Supermarket).
    \item \textbf{Item Normalization Node (normalize\_items\_node):} Unifies unit representations across stores (e.g., converting grams to kilograms, and matching brand variations).
    \item \textbf{Basket Optimization Node (optimize\_basket\_node):} Computes total costs for each individual store ($C_{\text{single}}$) and calculates the optimal cross-store partition ($C_{\text{multi}}$) by greedily picking the lowest price per item while adding delivery fees ($F_s$) for each distinct store visited.
    \item \textbf{Proposal Generation Node (generate\_proposal\_node):} Assembles the final structured response containing the Store Journey Audit, itemized picking breakdown per vendor, and proof of net savings ($\Delta_{\text{savings}}$).
\end{enumerate}

Algorithm~\ref{alg:langgraph_pipeline} defines the end-to-end execution loop of the compiled LangGraph workflow:

\begin{algorithm}[H]
\caption{Autonomous Multi-Store LangGraph Decision Pipeline}
\label{alg:langgraph_pipeline}
\begin{algorithmic}[1]
\Require User natural language shopping goal $Q_{\text{user}}$, Registered store adapters $\mathcal{S}$
\Ensure Audited multi-store picking itinerary $\mathcal{P}$ with net savings proof $\Delta_{\text{savings}}$
\State $\text{state}.\text{requirements} \gets \textsc{ExtractRequirements}(Q_{\text{user}})$ \Comment{Node 1: LLM Pydantic parser}
\State $\text{state}.\text{requirements} \gets \textsc{ValidateAndEnrich}(\text{state}.\text{requirements})$ \Comment{Node 2: Default staple fallback}
\State $\mathcal{C} \gets \emptyset$
\For{each required item $it \in \text{state}.\text{requirements}.\text{items}$} \Comment{Node 3: Parallel store discovery}
    \State $\text{candidates} \gets \textsc{ParallelSearchAcrossStores}(\mathcal{S}, it.\text{name}, it.\text{category})$
    \State $\mathcal{C} \gets \mathcal{C} \cup \text{candidates}$
\EndFor
\State $\mathcal{C}_{\text{norm}} \gets \textsc{NormalizeUnitsAndPrices}(\mathcal{C})$ \Comment{Node 4: Metric standardization}
\State $C_{\text{single}} \gets \min_{s \in \mathcal{S}} \left( \sum_{i} P_{s, i} + F_s \right)$ \Comment{Node 5: Single-store baseline}
\State $C_{\text{multi}} \gets \sum_{s \in \mathcal{S}_{\text{visited}}} \left( \sum_{i \in I_s} \min_{s'} P_{s', i} + F_s \right)$ \Comment{Multi-store partition}
\State $\Delta_{\text{savings}} \gets C_{\text{single}} - C_{\text{multi}}$
\State $\mathcal{P} \gets \textsc{CompileStoreJourneyAudit}(\mathcal{C}_{\text{norm}}, C_{\text{single}}, C_{\text{multi}}, \Delta_{\text{savings}})$ \Comment{Node 6: Visual itinerary}
\State \Return $\mathcal{P}$
\end{algorithmic}
\end{algorithm}

\section{Transaction Management and Atomic Quotation Acceptance}

To prevent race conditions during reverse RFQ negotiations, Algorithm~\ref{alg:atomic_acceptance} details the atomic multi-document state transition implemented in \texttt{backend/services/quotationEngine.js}:

\begin{algorithm}[H]
\caption{Atomic Concurrency-Controlled Quotation Acceptance}
\label{alg:atomic_acceptance}
\begin{algorithmic}[1]
\Require Request ID $id_{\text{req}}$, Selected Quotation ID $id_{\text{quote}}$, Customer ID $id_{\text{cust}}$
\Ensure Confirmed Order record $O$, Real-time WebSocket event dispatch
\State $Q \gets \textsc{FindQuotationById}(id_{\text{quote}})$
\If{$Q = \text{null}$ \textbf{or} $t_{\text{now}} > Q.\text{expiresAt}$ \textbf{or} $Q.\text{status} \ne \text{\texttt{'open'}}$}
    \State \textbf{throw} \text{Error(``Quotation is expired or no longer actionable'')}
\EndIf
\State $R \gets \textsc{FindBuyerRequestById}(id_{\text{req}})$
\If{$R.\text{status} = \text{\texttt{'accepted'}}$}
    \State \textbf{throw} \text{Error(``Request already converted into an active order'')}
\EndIf
\State \textbf{begin atomic transaction}
\State $Q.\text{status} \gets \text{\texttt{'accepted'}}$
\State $\textsc{RejectOtherQuotations}(id_{\text{req}}, id_{\text{quote}})$ \Comment{Set status to \texttt{'rejected'} for competing bids}
\State $O \gets \textsc{CreateOrderDocument}(id_{\text{cust}}, Q.\text{store}, Q.\text{lineItems}, Q.\text{grandTotal})$
\State $R.\text{status} \gets \text{\texttt{'accepted'}}$
\State $R.\text{convertedOrderId} \gets O.\text{\_id}$
\State \textbf{commit transaction}
\State $\textsc{EmitSocketEvent}(\texttt{rfq\_} + id_{\text{req}}, \text{\texttt{'quote\_update'}}, Q)$
\State $\textsc{EmitSocketEvent}(\texttt{order\_} + O.\text{\_id}, \text{\texttt{'order\_status\_change'}}, O)$
\State \Return $O$
\end{algorithmic}
\end{algorithm}

\section{Real-Time WebSocket Event Multiplexing}

The real-time layer is managed in \texttt{backend/server.js} using Socket.IO. To prevent event cross-talk between unrelated users, connections are partitioned into isolated rooms:

\begin{itemize}
    \item \textbf{Merchant RFQ Room (\texttt{merchants\_rfq}):} All authenticated merchants join this broadcast channel upon login. When any buyer posts a new requirement, the server emits \texttt{new\_rfq\_available} with summary details (delivery locality, item count, target deadline).
    \item \textbf{Request-Specific Negotiation Room (\texttt{rfq\_<id>}):} The buyer and participating merchants join a scoped room for that specific RFQ. When a merchant submits or revises an itemized quotation, or when the buyer submits a counter-offer, the server emits \texttt{quote\_update} exclusively to room participants.
    \item \textbf{Live Order Tracking Room (\texttt{order\_<id>}):} When an order is placed, both the customer and fulfilling merchant join the order room. Status updates (\texttt{placed} $\to$ \texttt{confirmed} $\to$ \texttt{out\_for\_delivery} $\to$ \texttt{delivered}) are broadcast instantaneously, triggering live UI badge updates without browser refreshes.
\end{itemize}

\section{Security Architecture and Sensitive Data Protection}

Security and privacy controls are enforced across all layers of the GroDealz application:

\begin{enumerate}
    \item \textbf{Cryptographic Authentication Middleware (\texttt{auth.js}):} Every protected route intercepts the HTTP \texttt{Authorization} header, extracts the Bearer JWT token, verifies its cryptographic signature using HMAC-SHA256, and attaches the decoded user payload to Express \texttt{req.user}. Requests with expired or tampered tokens are rejected with HTTP 401 Unauthorized.
    \item \textbf{Credential Protection:} User passwords are encrypted prior to database insertion using \texttt{bcryptjs} with 10 salt rounds. Plaintext passwords are never logged or stored.
    \item \textbf{Automated PII Sanitization Pipeline:} In the Customer Support CRM (\texttt{backend/routes/support.js}), Algorithm~\ref{alg:pii_masking} scrubs sensitive customer contact information and delivery landmarks before intent classification:
\end{enumerate}

\begin{algorithm}[H]
\caption{Automated Sensitive PII Masking and Ticket Triage (\texttt{support.js})}
\label{alg:pii_masking}
\begin{algorithmic}[1]
\Require Support ticket text $T$, Masking toggle flag $F_{\text{mask}}$
\Ensure Sanitized ticket text $T_{\text{clean}}$, Predicted Intent category $C$, Confidence score $\sigma$
\If{$F_{\text{mask}} = \text{\textbf{true}}$}
    \State $\text{pat}_{\text{phone}} \gets \text{\texttt{/(+91)?[6-9]\textbackslash d\{9\}/}}$ \Comment{Regex matching 10-digit mobile numbers}
    \State $\text{pat}_{\text{email}} \gets \text{\texttt{/[a-zA-Z0-9.\_\%+-]+@[a-zA-Z0-9.-]+\textbackslash.[a-zA-Z]\{2,\}/}}$
    \State $T_{\text{clean}} \gets \textsc{RegexReplace}(T, \text{pat}_{\text{phone}}, \text{\texttt{'[PHONE\_REDACTED]'}})$
    \State $T_{\text{clean}} \gets \textsc{RegexReplace}(T_{\text{clean}}, \text{pat}_{\text{email}}, \text{\texttt{'[EMAIL\_REDACTED]'}})$
\Else
    \State $T_{\text{clean}} \gets T$
\EndIf
\State $\langle C, \sigma \rangle \gets \textsc{ClassifyIntentWithNLP}(T_{\text{clean}})$ \Comment{Identifies issue: delivery, item, payment}
\State \Return $\langle T_{\text{clean}}, C, \sigma \rangle$
\end{algorithmic}
\end{algorithm}
